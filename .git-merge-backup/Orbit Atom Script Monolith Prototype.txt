CREATE OR REPLACE FUNCTION public.process_forensic_audit() RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO public.audit_logs (user_id, action_type, table_name, record_id, old_data, new_data)
    VALUES (auth.uid(), TG_OP, TG_TABLE_NAME, COALESCE(NEW.id, OLD.id), to_jsonb(OLD), to_jsonb(NEW));
    RETURN NEW;
END; $$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION public.set_updated_at() RETURNS TRIGGER AS $$
BEGIN NEW.updated_at = clock_timestamp(); RETURN NEW; END; $$ LANGUAGE plpgsql;

-- B. Trip Lead Clearance Gate
CREATE OR REPLACE FUNCTION public.check_trip_lead_clearance() RETURNS TRIGGER AS $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM public.staff_profiles 
        WHERE id = NEW.lead_satellite_id 
        AND (role IN ('Satellite', 'Teacher', 'Command', 'Central'))
    ) AND NEW.lead_satellite_id IS NOT NULL THEN
        RAISE EXCEPTION 'AUTHORITY_ERROR: Assigned Lead must have Teacher or Satellite clearance.';
    END IF;
    RETURN NEW;
END; $$ LANGUAGE plpgsql;

-- C. Dynamic Engine Attachments
DO $$ 
DECLARE t TEXT;
BEGIN
    -- Attach updated_at triggers universally (Filtering for BASE TABLES only)
    FOR t IN SELECT table_name FROM information_schema.tables WHERE table_schema = 'public' AND table_type = 'BASE TABLE'
    LOOP
        -- Only attach if the column actually exists
        IF EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema = 'public' AND table_name = t AND column_name = 'updated_at') THEN
            EXECUTE format('CREATE TRIGGER trig_set_updated_%I BEFORE UPDATE ON public.%I FOR EACH ROW EXECUTE FUNCTION public.set_updated_at()', t, t);
        END IF;
    END LOOP;

    -- Attach Forensic Black Box triggers universally (Excluding logs and PostGIS tables)
    FOR t IN SELECT table_name FROM information_schema.tables 
    WHERE table_schema = 'public' 
    AND table_type = 'BASE TABLE'
    AND table_name NOT IN ('audit_logs', 'bus_telemetry_logs', 'spatial_ref_sys')
    LOOP
        EXECUTE format('CREATE TRIGGER trig_audit_%I AFTER INSERT OR UPDATE OR DELETE ON public.%I FOR EACH ROW EXECUTE FUNCTION public.process_forensic_audit()', t, t);
    END LOOP;
END $$;

-- ==========================================
-- 1. THE VIP BOUNCERS (Security Definer Functions)
-- ==========================================
-- These functions bypass RLS to quickly check a user's role/district without infinite looping.
CREATE OR REPLACE FUNCTION public.get_my_role() RETURNS TEXT AS $$
    SELECT role FROM public.staff_profiles WHERE id = auth.uid() LIMIT 1;
$$ LANGUAGE sql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION public.get_my_district() RETURNS UUID AS $$
    SELECT district_id FROM public.staff_profiles WHERE id = auth.uid() LIMIT 1;
$$ LANGUAGE sql SECURITY DEFINER;

-- ==========================================
-- 2. PARENT POLICIES (The Guardian Gate)
-- ==========================================
-- Parents can only see themselves.
DROP POLICY IF EXISTS parent_view_self ON public.parents;
CREATE POLICY parent_view_self ON public.parents FOR SELECT USING (id = auth.uid());

-- Parents can update their own profile (except verification status).
DROP POLICY IF EXISTS parent_update_self ON public.parents;
CREATE POLICY parent_update_self ON public.parents FOR UPDATE USING (id = auth.uid());

-- Parents can only see students legally linked to them in the guardians table.
DROP POLICY IF EXISTS parent_view_own_students ON public.students;
CREATE POLICY parent_view_own_students ON public.students FOR SELECT

-- ORBIT INTELLIGENCE: AUTOMATIC GROUNDING ENGINE
-- Purpose: Instantly grounds a bus if a critical safety failure is logged.

-- 1. THE GROUNDING LOGIC FUNCTION
CREATE OR REPLACE FUNCTION public.fn_auto_ground_bus()
RETURNS TRIGGER AS $$
BEGIN
    -- If the inspection is marked as 'Failed' or 'Critical'
    IF (NEW.inspection_status IN ('Failed', 'Critical', 'Safety_Pull')) THEN
        
        -- Update the physical bus status
        UPDATE public.buses
        SET 
            is_grounded = true,
            status = 'Maintenance_Required',
            updated_at = clock_timestamp()
        WHERE id = NEW.bus_id;

        -- Create an automatic high-priority Maintenance Work Order
        INSERT INTO public.maintenance_work_orders (
            bus_id,
            issue_description,
            priority,
            status
        ) VALUES (
            NEW.bus_id,
            'AUTO-GENERATED: Bus grounded due to failed inspection. See Inspection ID: ' || NEW.id,
            'Urgent',
            'Open'
        );

        RAISE NOTICE 'BUS GROUNDED: Automatic safety interlock engaged for Bus %', NEW.bus_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 2. THE TRIP-WIRE TRIGGER
-- This fire every time a new inspection is filed.
DROP TRIGGER IF EXISTS trig_auto_grounding ON public.bus_inspections;
CREATE TRIGGER trig_auto_grounding
AFTER INSERT OR UPDATE OF inspection_status ON public.bus_inspections
FOR EACH ROW
EXECUTE FUNCTION public.fn_auto_ground_bus();

-- B. SMART BOARDING (With Audible Metadata)
CREATE OR REPLACE FUNCTION public.fn_student_board_bus(p_student_id UUID, p_bus_id UUID)
RETURNS JSONB AS $$
DECLARE
    v_is_on_manifest BOOLEAN;
    v_is_at_school BOOLEAN;
    v_trip_id UUID;
BEGIN
    -- Identify trip if applicable
    SELECT trip_id INTO v_trip_id FROM public.trip_bus_assignments WHERE bus_id = p_bus_id AND handshake_status = true LIMIT 1;

    -- Check school registry
    SELECT EXISTS (SELECT 1 FROM public.students WHERE id = p_student_id) INTO v_is_at_school;

    -- Check current manifest
    SELECT EXISTS (
        SELECT 1 FROM public.halo_manifest_snapshots 
        WHERE student_id = p_student_id AND current_bus_id = p_bus_id
    ) INTO v_is_on_manifest;

    IF v_is_on_manifest THEN
        UPDATE public.halo_manifest_snapshots 
        SET status = 'Boarded', last_seen_at = now()
        WHERE student_id = p_student_id AND current_bus_id = p_bus_id;
        RETURN jsonb_build_object('status', 'Success', 'audible', false);
    ELSIF v_is_at_school THEN
        INSERT INTO public.emergency_flares (bus_id, flare_type, severity, metadata)
        VALUES (p_bus_id, 'No_Student_Flag', 'High', '{"is_audible": true, "alert_tone": "vocal_warning"}');
        RETURN jsonb_build_object('status', 'Flagged', 'message', 'Missing Slip', 'audible', true);
    ELSE
        INSERT INTO public.emergency_flares (bus_id, flare_type, severity, metadata)
        VALUES (p_bus_id, 'Wrong_Bus_Flag', 'Critical', '{"is_audible": true, "alert_tone": "siren"}');
        RETURN jsonb_build_object('status', 'Flagged', 'message', 'Wrong Bus', 'audible', true);
    END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 3. The Execution Engine (The Digital Handshake & Alert Router)
CREATE OR REPLACE FUNCTION public.fn_approve_parent_checkout(p_checkout_id UUID)
RETURNS JSONB AS $$
DECLARE
    v_checkout RECORD;
    v_is_legal_guardian BOOLEAN;
    v_approver_role TEXT;
    v_assigned_satellite_id UUID;
    v_lead_satellite_id UUID;
    v_requesting_parent_name TEXT;
BEGIN
    -- Get the request details
    SELECT * INTO v_checkout FROM public.field_trip_checkouts WHERE id = p_checkout_id;
    
    -- Identify the Approver's Role
    SELECT role INTO v_approver_role FROM public.staff_profiles WHERE id = auth.uid();

    -- Identify the specifically assigned Satellite (Teacher) and the Lead Satellite
    SELECT satellite_id INTO v_assigned_satellite_id FROM public.trip_chaperone_groups 
    WHERE trip_id = v_checkout.trip_id AND node_id = v_checkout.requesting_parent_node_id LIMIT 1;

    SELECT lead_id INTO v_lead_satellite_id FROM public.trips WHERE id = v_checkout.trip_id;

    -- Get the name of the parent checking the student out (for the co-parent alert)
    SELECT COALESCE(first_name, '') || ' ' || COALESCE(last_name, '') INTO v_requesting_parent_name 
    FROM public.guardian_profiles WHERE id = v_checkout.requesting_parent_node_id;

    -- STEP 1: Verify Approver Authority
    IF v_approver_role NOT IN ('Satellite', 'Lead_Satellite', 'Command', 'Principal') THEN
        RAISE EXCEPTION 'Authority Error: Invalid role for checkout approval.';
    END IF;

    -- STRICT GATE: If the approver is a standard Teacher, they MUST be the specific assigned Teacher.
    IF v_approver_role = 'Satellite' AND auth.uid() != v_assigned_satellite_id THEN
        RAISE EXCEPTION 'Authority Error: Only the assigned Teacher or Trip Lead can approve this checkout.';
    END IF;

    -- STEP 2: Verify Legal Guardianship
    SELECT EXISTS (
        SELECT 1 FROM public.student_guardians 
        WHERE student_id = v_checkout.student_id 
        AND parent_id = v_checkout.requesting_parent_node_id
    ) INTO v_is_legal_guardian;

    IF NOT v_is_legal_guardian THEN
        UPDATE public.field_trip_checkouts SET status = 'Denied', resolved_at = now() WHERE id = p_checkout_id;
        RAISE EXCEPTION 'Security Error: This Chaperone is not a legally registered guardian for this student.';
    END IF;

    -- STEP 3: Execute the Checkout Logic
    
    -- A. Update the checkout log
    UPDATE public.field_trip_checkouts 
    SET status = 'Approved', approving_satellite_id = auth.uid(), resolved_at = now() 
    WHERE id = p_checkout_id;

    -- B. Remove the student from the return bus expected manifest
    UPDATE public.trip_manifest 
    SET expected_status = 'EXCUSED_CHECKOUT' 
    WHERE trip_id = v_checkout.trip_id AND student_id = v_checkout.student_id;

    -- C. Update the student's global status
    UPDATE public.students 
    SET current_status = 'off_bus', status_note = 'Checked_Out_By_Parent'
    WHERE id = v_checkout.student_id;

    -- ==========================================
    -- STEP 4: ALERT ROUTING MATRIX
    -- ==========================================

    -- ALERT 1: PILOT (The "What" - Operational)
    INSERT INTO public.outbound_alerts (recipient_id, alert_type, priority, title, body)
    SELECT pilot_id, 'Routine_Tap', 'Normal', 'MANIFEST UPDATE', 
           'Student has been checked out by parent. Removed from return manifest.'
    FROM public.trip_bus_assignments WHERE trip_id = v_checkout.trip_id;

    -- ALERT 2: LEAD SATELLITE (The "Why" - Tactical)
    IF v_lead_satellite_id IS NOT NULL THEN
        INSERT INTO public.outbound_alerts (recipient_id, alert_type, priority, title, body)
        VALUES (
            v_lead_satellite_id, 'Rescue_Update', 'High', 'MASTER MANIFEST UPDATE', 
            'Student securely checked out by ' || v_requesting_parent_name || '.'
        );
    END IF;

    -- ALERT 3: CENTRAL / FRONT OFFICE (The "Why" - Administrative)
    INSERT INTO public.outbound_alerts (recipient_id, alert_type, priority, title, body)
    SELECT sp.id, 'Rescue_Update', 'Normal', 'ADMIN RECORD: FIELD TRIP CHECKOUT', 
           'Student safely checked out by ' || v_requesting_parent_name || '. Master Manifest adjusted.'
    FROM public.staff_profiles sp
    WHERE sp.role = 'Central' 
    AND sp.school_id = (SELECT school_id FROM public.students WHERE id = v_checkout.student_id);

    -- ALERT 4: CO-PARENTS / ADDITIONAL GUARDIANS (Transparency)
    INSERT INTO public.outbound_alerts (recipient_id, alert_type, priority, title, body)
    SELECT parent_id, 'Routine_Tap', 'High', 'STUDENT CHECKOUT NOTIFICATION', 
           'Your student was checked out from the field trip by ' || v_requesting_parent_name || '.'
    FROM public.student_guardians
    WHERE student_id = v_checkout.student_id 
    AND parent_id != v_checkout.requesting_parent_node_id; -- Do not ping the parent who just made the request

    RETURN jsonb_build_object('status', 'Success', 'message', 'Student safely checked out. Manifests and alerts routed.');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION public.fn_archive_stale_data()
RETURNS VOID AS $$
BEGIN
    -- Move Telemetry older than 30 days
    WITH moved_telemetry AS (
        DELETE FROM public.bus_telemetry_logs
        WHERE synced_at < (now() - interval '30 days')
        RETURNING *
    )
    INSERT INTO archive.bus_telemetry_logs SELECT * FROM moved_telemetry;

    -- Move Audit Logs older than 90 days (Compliance requirement)
    WITH moved_audits AS (
        DELETE FROM public.audit_logs
        WHERE created_at < (now() - interval '90 days')
        RETURNING *
    )
    INSERT INTO archive.audit_logs SELECT * FROM moved_audits;

    -- Hard Delete Telemetry older than 1 year from the archive (Save storage costs)
    DELETE FROM archive.bus_telemetry_logs WHERE synced_at < (now() - interval '1 year');
    
    RAISE NOTICE 'Chronos Janitor Engine completed cycle.';
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ==========================================
-- 4. THE DATA LOCKBOX ENGINE (Automated Purge Trigger)
-- Purpose: Instantly strips sensitive operational data upon trip completion
-- ==========================================
CREATE OR REPLACE FUNCTION public.fn_execute_trip_lockbox_purge()
RETURNS TRIGGER AS $$
DECLARE
    v_halo_session_id UUID;
BEGIN
    -- Only engage the lockbox when a trip is officially finalized
    IF (NEW.status IN ('Completed', 'Canceled')) THEN
        
        -- Find the Halo session associated with this trip via assignments
        SELECT hs.id INTO v_halo_session_id 
        FROM public.halo_sessions hs
        JOIN public.trip_bus_assignments tba ON hs.bus_id = tba.bus_id
        WHERE tba.trip_id = NEW.id LIMIT 1;

        -- STEP 1: Flush the Halo Manifest Snapshot to Cold Storage Archive
        IF v_halo_session_id IS NOT NULL THEN
            WITH flushed_snapshots AS (
                DELETE FROM public.halo_manifest_snapshots
                WHERE halo_session_id = v_halo_session_id
                RETURNING *
            )
            INSERT INTO archive.halo_manifest_snapshots SELECT * FROM flushed_snapshots;
            
            -- Close out the live Halo session
            UPDATE public.halo_sessions 
            SET bailout_status = 'Rescued_Complete', updated_at = now()
            WHERE id = v_halo_session_id;
        END IF;

        -- STEP 2: Flush In-App Chat Logs for the Trip Radio Channel
        WITH flushed_chat AS (
            DELETE FROM public.comms_messages
            WHERE channel_id IN (SELECT id FROM public.comms_channels WHERE trip_id = NEW.id)
            RETURNING *
        )
        INSERT INTO archive.comms_messages SELECT * FROM flushed_chat;

        -- STEP 3: Flush Parent-Chaperone Field Trip Checkouts
        WITH flushed_checkouts AS (
            DELETE FROM public.field_trip_checkouts
            WHERE trip_id = NEW.id
            RETURNING *
        )
        INSERT INTO archive.field_trip_checkouts SELECT * FROM flushed_checkouts;

        -- STEP 4: Soft-Delete/Archive Active Manifest
        UPDATE public.trip_manifest 
        SET expected_status = 'EXCUSED'
        WHERE trip_id = NEW.id;

        -- Log the cryptographic scrub event for Government Compliance Audits
        INSERT INTO public.incident_dispatch_logs (action_taken, dispatch_notes)
        VALUES (
            'DATA_LOCKBOX_SECURED', 
            'Trip ID ' || NEW.id || ' finalized. Active manifests, chat protocols, and student checkout data scrubbed from live operational layers.'
        );

    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 5. ATTACHING THE TRIPWIRE TRIGGER

DROP TRIGGER IF EXISTS trig_lockbox_purge ON public.trips;

CREATE OR REPLACE FUNCTION public.handle_graduation()
RETURNS TRIGGER AS $$
BEGIN
  -- If student status changes to 'graduated'
  IF NEW.status = 'graduated' THEN
    -- 1. Remove the Pin-specific role if it exists
    UPDATE public.profiles 
    SET preferred_app = 'trust' -- Force the default to Trust
    WHERE id = NEW.user_id;
    
    -- 2. Optionally, nullify the student_id to "deactivate" the Pin app view
    UPDATE public.profiles
    SET student_id = NULL
    WHERE id = NEW.user_id;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

    -- We look in 'authorized_relationships' to see if they are a guardian/adult

      SELECT 1 
      FROM public.authorized_relationships

    ) THEN
      -- Only if they HAVE a dual-role, promote them to 'trust'
      UPDATE public.profiles
      SET preferred_app = 'trust'
      WHERE id = NEW.user_id;

      -- Optional: If they DON'T have a dual role, you might want to 
      -- set them to 'archived' or just leave them alone. 
      -- Here we just do nothing, keeping them as 'point'.
      NULL; 
    END IF;
    
  END IF;
  RETURN NEW;
END;

  UPDATE public.profiles

  WHERE id = auth.uid();
  
  -- OR, if this is a parent, you would insert into authorized_relationships here.
  
  RETURN true;
END;

-- ==========================================
-- 2. THE HANDSHAKE ENFORCEMENT ENGINE
-- ==========================================
CREATE OR REPLACE FUNCTION public.fn_process_chaperone_handshake(
    p_handshake_id UUID,
    p_action TEXT, -- 'Approved' or 'Rejected'
    p_target_satellite_id UUID DEFAULT NULL -- Optional reassignment by Lead
)
RETURNS JSONB AS $$
DECLARE
    v_handshake RECORD;
    v_approver_id UUID;
    v_approver_role TEXT;
    v_final_satellite_id UUID;
BEGIN
    v_approver_id := auth.uid();

    -- Fetch the pending handshake request
    SELECT * INTO v_handshake FROM public.trip_chaperone_handshakes 
    WHERE id = p_handshake_id AND handshake_status = 'Pending';

    IF v_handshake.id IS NULL THEN
        RAISE EXCEPTION 'Handshake Error: Request not found or already processed.';
    END IF;

    -- Fetch the teacher's role within the district
    SELECT role INTO v_approver_role FROM public.staff_profiles 
    WHERE id = v_approver_id;

    -- SECURITY GATE: Only valid trip staff can process handshakes
    IF v_approver_role NOT IN ('Satellite', 'Lead_Satellite', 'Staff_Teacher') THEN
        RAISE EXCEPTION 'Security Exception: Unauthorized staff signature.';
    END IF;

    IF p_action = 'Rejected' THEN
        UPDATE public.trip_chaperone_handshakes 
        SET handshake_status = 'Rejected', resolved_at = clock_timestamp()
        WHERE id = p_handshake_id;
        RETURN jsonb_build_object('status', 'Rejected', 'message', 'Handshake declined.');
    END IF;

    -- ENFORCE AUTHORITY RULE MATRIX
    IF v_approver_role = 'Lead_Satellite' THEN
        -- Lead Satellites have full delegation authority
        IF p_target_satellite_id IS NOT NULL THEN
            v_final_satellite_id := p_target_satellite_id; -- Reassigned to another teacher
        ELSE
            v_final_satellite_id := v_handshake.requested_satellite_id; -- Left as original
        END IF;
    ELSE
        -- Standard Satellites CANNOT reassign. They can only accept for themselves.
        IF p_target_satellite_id IS NOT NULL AND p_target_satellite_id != v_approver_id THEN
            RAISE EXCEPTION 'Privilege Violation: Standard Satellites do not possess delegation authority.';
        END IF;
        v_final_satellite_id := v_approver_id;
    END IF;

    -- UPDATE THE LIVE DEPLOYMENT RECORDS
    UPDATE public.trip_chaperone_handshakes 
    SET handshake_status = 'Approved', 
        assigned_satellite_id = v_final_satellite_id,
        resolved_at = clock_timestamp()
    WHERE id = p_handshake_id;

    -- Inject chaperone directly into the active trip manifest operational group
    INSERT INTO public.trip_chaperone_groups (trip_id, chaperone_id, supervisor_satellite_id)
    VALUES (v_handshake.trip_id, v_handshake.guardian_id, v_final_satellite_id)
    ON CONFLICT (trip_id, chaperone_id) 
    DO UPDATE SET supervisor_satellite_id = v_final_satellite_id;

    RETURN jsonb_build_object(
        'status', 'Success', 
        'assigned_to', v_final_satellite_id,
        'message', 'Chaperone successfully bound and synchronized to group.'
    );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

COMMIT;

-- 2. CHRONOS RETENTION ADJUSTMENT (3-Year Legal Minimum)
-- ==========================================
CREATE OR REPLACE FUNCTION public.fn_archive_stale_data()
RETURNS VOID AS $$
BEGIN
    -- Move Telemetry older than 30 days to cold storage
    WITH moved_telemetry AS (
        DELETE FROM public.bus_telemetry_logs
        WHERE synced_at < (now() - interval '30 days')
        RETURNING *
    )
    INSERT INTO archive.bus_telemetry_logs SELECT * FROM moved_telemetry;

    -- Move Audit Logs older than 90 days to cold storage
    WITH moved_audits AS (
        DELETE FROM public.audit_logs
        WHERE created_at < (now() - interval '90 days')
        RETURNING *
    )
    INSERT INTO archive.audit_logs SELECT * FROM moved_audits;

    -- COMPLIANCE UPDATE: Hard Delete Telemetry older than 3 years (Not 1 year)
    DELETE FROM archive.bus_telemetry_logs WHERE synced_at < (now() - interval '3 years');
    
    -- COMPLIANCE UPDATE: Audit logs must be retained for 7 years for potential litigation
    DELETE FROM archive.audit_logs WHERE created_at < (now() - interval '7 years');
    
    RAISE NOTICE 'Chronos Janitor Engine completed cycle under 3/7 year compliance rules.';
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

COMMIT;

ADD COLUMN IF NOT EXISTS route_status TEXT DEFAULT 'Idle' CHECK (route_status IN ('Idle', 'En_Route', 'Arriving', 'At_Stop', 'Completed'));

-- 3. THE "NO CHILD LEFT BEHIND" SAFEGUARD
-- This function checks for boarded students before allowing a route switch.
CREATE OR REPLACE FUNCTION public.fn_check_bus_clearance(p_bus_id UUID)
RETURNS BOOLEAN AS $$
DECLARE
    v_still_onboard INT;
BEGIN
    SELECT COUNT(*) INTO v_still_onboard 
    FROM public.halo_manifest_snapshots 
    WHERE current_bus_id = p_bus_id AND status = 'Boarded';

    RETURN (v_still_onboard = 0);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 4. THE AUTOMATED GPS ROUTE ENGINE
-- Updates location and handles the Closing In -> Arrived -> Departed cycle.
CREATE OR REPLACE FUNCTION public.fn_process_route_gps(
    p_bus_id UUID,
    p_lat DOUBLE PRECISION,
    p_lng DOUBLE PRECISION
)
RETURNS JSONB AS $$
DECLARE
    v_curr_route UUID;
    v_next_stop RECORD;
    v_dist DOUBLE PRECISION;
    v_is_clear BOOLEAN;
    v_event TEXT := 'Navigating';
BEGIN
    SELECT current_route_id INTO v_curr_route FROM public.buses WHERE id = p_bus_id;

    -- Find the next stop in the sequence that we haven't reached yet
    SELECT * INTO v_next_stop FROM public.route_stops 
    WHERE route_id = v_curr_route 
    AND sequence_order > (SELECT COALESCE(sequence_order, 0) FROM public.route_stops WHERE id = (SELECT last_stop_id FROM public.buses WHERE id = p_bus_id))
    ORDER BY sequence_order ASC LIMIT 1;

    v_dist := ST_Distance(v_next_stop.geofence_center, ST_SetSRID(ST_MakePoint(p_lng, p_lat), 4326)::geography);

    -- LOGIC: STATUS TRANSITIONS
    IF v_dist <= 500 AND v_dist > 50 THEN
        v_event := 'Closing_In'; -- Pings Parents/Students for countdown timer
    ELSIF v_dist <= 50 THEN
        v_event := 'Arrived'; -- Update Bus Status to 'At_Stop'
        UPDATE public.buses SET route_status = 'At_Stop', last_stop_id = v_next_stop.id WHERE id = p_bus_id;
    END IF;

    -- LOGIC: ROUTE COMPLETION & SWITCHING
    -- If no more stops remain and we are back at school
    IF v_next_stop IS NULL THEN
        v_is_clear := public.fn_check_bus_clearance(p_bus_id);
        
        IF NOT v_is_clear THEN
            -- AUDIBLE & VISUAL ALERT: Triggered for the Pilot
            INSERT INTO public.emergency_flares (bus_id, flare_type, severity, status, metadata)
            VALUES (p_bus_id, 'Student_Still_Onboard', 'Critical', 'Pending', '{"is_audible": true, "alert_tone": "vocal_warning"}');
            RETURN jsonb_build_object('event', 'STATIONARY_CHECK_REQUIRED', 'error', 'Students Still Onboard');
        ELSE
            -- SUCCESS: Automatic Route Flip
            UPDATE public.buses 
            SET current_route_id = next_route_id, 
                next_route_id = NULL,
                last_stop_id = NULL,
                route_status = 'Idle'
            WHERE id = p_bus_id;
            v_event := 'Route_Completed_Switched';
        END IF;
    END IF;

    RETURN jsonb_build_object('event', v_event, 'dist_to_next', v_dist, 'stop_name', v_next_stop.stop_name);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

COMMIT;

-- 2. THE GPS INTELLIGENCE FUNCTION
-- This is called by the Pilot's tablet every few seconds.
-- It updates the bus location and checks if it has crossed the "Event Horizon."
CREATE OR REPLACE FUNCTION public.fn_update_bus_location_with_geofence(
    p_bus_id UUID,
    p_lat DOUBLE PRECISION,
    p_lng DOUBLE PRECISION
)
RETURNS JSONB AS $$
DECLARE
    v_school_id UUID;
    v_center GEOGRAPHY(POINT, 4326);
    v_radius INT;
    v_current_dist DOUBLE PRECISION;
    v_was_on_campus BOOLEAN;
    v_is_now_on_campus BOOLEAN;
BEGIN
    -- 1. Get current status and school geofence data
    SELECT b.school_id, b.is_on_campus, s.geofence_center, s.geofence_radius_meters 
    INTO v_school_id, v_was_on_campus, v_center, v_radius
    FROM public.buses b
    JOIN public.schools s ON b.school_id = s.id
    WHERE b.id = p_bus_id;

    -- 2. Update the physical location in the database
    UPDATE public.buses 
    SET current_location = ST_SetSRID(ST_MakePoint(p_lng, p_lat), 4326)::geography
    WHERE id = p_bus_id;

    -- 3. Calculate distance from the school center
    v_current_dist := ST_Distance(v_center, ST_SetSRID(ST_MakePoint(p_lng, p_lat), 4326)::geography);

    -- 4. Determine if we are still within the fence
    v_is_now_on_campus := (v_current_dist <= v_radius);

    -- 5. TRIGGER: The "Event Horizon" Crossing
    -- If we WERE on campus but ARE NOT anymore, lock the manifest.
    IF v_was_on_campus AND NOT v_is_now_on_campus THEN
        UPDATE public.buses SET is_on_campus = false WHERE id = p_bus_id;
        
        -- LOG THE DEPARTURE FOR COMMAND
        INSERT INTO public.outbound_alerts (recipient_id, alert_type, priority, title, body)
        SELECT id, 'System_Log', 'Medium', 'GEOFENCE BREACH: DEPARTED', 
               'Bus ' || p_bus_id || ' has exited the school perimeter. Manifest is now LOCKED.'
        FROM public.staff_profiles WHERE role IN ('Command', 'Central');
        
        RETURN jsonb_build_object('status', 'Departed', 'lock_engaged', true, 'distance', v_current_dist);
    END IF;

    RETURN jsonb_build_object('status', CASE WHEN v_is_now_on_campus THEN 'On_Campus' ELSE 'In_Transit' END, 'lock_engaged', NOT v_is_now_on_campus, 'distance', v_current_dist);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

    END IF;

END;

RETURNS JSONB AS $$

    v_trip_id UUID;
    v_is_on_manifest BOOLEAN;
    v_is_at_school BOOLEAN;

    SELECT EXISTS (SELECT 1 FROM public.trips WHERE id = v_trip_id AND master_manifest @> jsonb_build_array(jsonb_build_object('id', p_student_id))) INTO v_is_on_manifest;
    SELECT EXISTS (SELECT 1 FROM public.students WHERE id = p_student_id) INTO v_is_at_school;

        RETURN jsonb_build_object('status', 'Success', 'audible', false);
    ELSIF v_is_at_school THEN
        INSERT INTO public.emergency_flares (bus_id, flare_type, severity, status, metadata)
        VALUES (p_bus_id, 'No_Student_Flag', 'High', 'Pending', '{"is_audible": true, "alert_tone": "vocal_warning"}');

    ELSE
        INSERT INTO public.emergency_flares (bus_id, flare_type, severity, status, metadata)
        VALUES (p_bus_id, 'Wrong_Bus_Flag', 'Critical', 'Pending', '{"is_audible": true, "alert_tone": "siren"}');
        RETURN jsonb_build_object('status', 'Flagged', 'audible', true);
    END IF;
END;

CREATE TABLE public.rescue_handshakes (

    halo_session_id UUID REFERENCES public.halo_sessions(id) ON DELETE CASCADE,
    source_bus_id UUID REFERENCES public.buses(id), -- The downed bus (if not evacuated)

    status TEXT DEFAULT 'Dispatched' CHECK (status IN ('Dispatched', 'In_Proximity', 'Requested', 'Accepted', 'Completed', 'Aborted')),

-- ==========================================
-- 2. THE PROXIMITY SENSOR (INTELLIGENCE)
-- ==========================================
-- This function monitors the Rescue Bus's GPS. 
-- When it gets within 50m of the target, it flips the status to 'In_Proximity'.
CREATE OR REPLACE FUNCTION public.fn_monitor_rescue_proximity()
RETURNS TRIGGER AS $$
DECLARE
    v_handshake_record RECORD;
    v_target_location GEOGRAPHY;
BEGIN
    -- Look for an active handshake involving this bus as the Rescue vehicle
    SELECT * INTO v_handshake_record 
    FROM public.rescue_handshakes 
    WHERE rescue_bus_id = NEW.bus_id AND status = 'Dispatched' 
    LIMIT 1;

    IF v_handshake_record.id IS NOT NULL THEN
        -- Determine if we are heading to a Bus or a Huddle
        IF v_handshake_record.handshake_type = 'Bus_to_Bus' THEN
            SELECT location_gps INTO v_target_location FROM public.bus_telemetry_logs 
            WHERE bus_id = v_handshake_record.source_bus_id ORDER BY created_at DESC LIMIT 1;
        ELSE
            SELECT current_huddle_location INTO v_target_location FROM public.halo_sessions 
            WHERE id = v_handshake_record.halo_session_id;
        END IF;

        -- Check 50-meter proximity
        IF ST_DWithin(NEW.location_gps, v_target_location, 50) THEN
            UPDATE public.rescue_handshakes 
            SET status = 'In_Proximity' 
            WHERE id = v_handshake_record.id;
            
            RAISE NOTICE 'RESCUE REACHED: Rescue Bus % is in range. Handshake available.', NEW.bus_id;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 3. THE BULK TRANSFER ENGINE
-- This moves the kids from the old manifest to the new one in a single transaction.
CREATE OR REPLACE FUNCTION public.fn_execute_manifest_transfer(p_handshake_id UUID)
RETURNS VOID AS $$
DECLARE
    v_handshake public.rescue_handshakes;
    v_halo_session public.halo_sessions;
BEGIN
    SELECT * INTO v_handshake FROM public.rescue_handshakes WHERE id = p_handshake_id;
    SELECT * INTO v_halo_session FROM public.halo_sessions WHERE id = v_handshake.halo_session_id;

    -- 1. Log the Exit from the old bus/halo
    INSERT INTO public.student_scan_events (student_id, bus_id, trip_id, scan_type, event_action, location_at_scan)
    SELECT 
        student_id, 
        v_handshake.source_bus_id, 
        (SELECT trip_id FROM public.bus_trip_handshakes WHERE bus_id = v_handshake.source_bus_id AND status = 'Active' LIMIT 1),
        'Manual_Pilot', 
        'Transfer', 
        (SELECT current_huddle_location FROM public.halo_sessions WHERE id = v_handshake.halo_session_id)
    FROM public.halo_manifest_snapshots 
    WHERE halo_session_id = v_handshake.halo_session_id AND is_verified = true;

    -- 2. Log the Entry into the Rescue Bus
    INSERT INTO public.student_scan_events (student_id, bus_id, trip_id, scan_type, event_action, location_at_scan)
    SELECT 
        student_id, 
        v_handshake.rescue_bus_id, 
        (SELECT trip_id FROM public.bus_trip_handshakes WHERE bus_id = v_handshake.rescue_bus_id AND status = 'Active' LIMIT 1),
        'Manual_Pilot', 
        'Boarded', 
        (SELECT current_huddle_location FROM public.halo_sessions WHERE id = v_handshake.halo_session_id)
    FROM public.halo_manifest_snapshots 
    WHERE halo_session_id = v_handshake.halo_session_id AND is_verified = true;

    -- 3. Close the Handshake
    UPDATE public.rescue_handshakes SET status = 'Completed', resolved_at = now() WHERE id = p_handshake_id;
    UPDATE public.halo_sessions SET bailout_status = 'Rescued_Complete' WHERE id = v_handshake.halo_session_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 4. ATTACH TRIGGER
DROP TRIGGER IF EXISTS trig_rescue_proximity ON public.bus_telemetry_logs;
CREATE TRIGGER trig_rescue_proximity
AFTER INSERT ON public.bus_telemetry_logs
FOR EACH ROW
EXECUTE FUNCTION public.fn_monitor_rescue_proximity();

COMMIT;

-- ============================================================================
-- ORBIT PARTITION 04: THE HALO PROTOCOL (BAILOUT & EVACUATION)
-- Purpose: Bus-to-Ground handoff, Manifest Dump, and Split Beacon Tracking
-- Status: Diamond Tier | Safety Critical
-- ============================================================================

BEGIN;

-- ==========================================
-- 1. THE HALO SESSION (THE LIFEBOAT)
-- ==========================================
CREATE TABLE public.halo_sessions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    flare_id UUID REFERENCES public.emergency_flares(id) ON DELETE CASCADE,
    bus_id UUID REFERENCES public.buses(id),
    pilot_id UUID REFERENCES auth.users(id),
    rescue_bus_id UUID REFERENCES public.buses(id), -- The incoming savior bus
    
    -- Split Beacon Tracking
    abandoned_bus_location GEOGRAPHY(POINT, 4326) NOT NULL,
    current_huddle_location GEOGRAPHY(POINT, 4326), -- Updated by Driver's device
    
    -- Headcount Logic
    expected_count INT NOT NULL,
    verified_count INT DEFAULT 0,
    bailout_status TEXT DEFAULT 'Evacuating' CHECK (bailout_status IN ('Evacuating', 'Huddled', 'Transferring', 'Rescued_Complete')),
    
    -- EMS Action Packet (JSON dump for first responders)
    ems_action_packet JSONB DEFAULT '{}'::jsonb,
    
    created_at TIMESTAMPTZ DEFAULT clock_timestamp(),
    updated_at TIMESTAMPTZ DEFAULT clock_timestamp()
);

-- ==========================================
-- 2. THE HALO MANIFEST (THE DATA DUMP)
-- ==========================================
CREATE TABLE public.halo_manifest_snapshots (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    halo_session_id UUID REFERENCES public.halo_sessions(id) ON DELETE CASCADE,
    student_id UUID REFERENCES public.students(id),
    verification_type TEXT CHECK (verification_type IN ('Tap', 'PIN', 'Visual_Manual')),
    is_verified BOOLEAN DEFAULT false,
    verified_at TIMESTAMPTZ
);

-- ==========================================
-- 3. THE HALO ENGINE (AUTOMATION)
-- ==========================================
CREATE OR REPLACE FUNCTION public.fn_initiate_halo_protocol()
RETURNS TRIGGER AS $$
DECLARE
    v_halo_id UUID;
    v_boarded_count INT;
BEGIN
    -- 1. Create the Halo Session
    INSERT INTO public.halo_sessions (
        flare_id,
        bus_id,
        pilot_id,
        abandoned_bus_location,
        current_huddle_location,
        expected_count
    ) VALUES (
        NEW.id,
        NEW.bus_id,
        NEW.triggered_by_id,
        NEW.initial_location,
        NEW.initial_location,
        (SELECT COUNT(*) FROM public.student_scan_events WHERE bus_id = NEW.bus_id AND event_action = 'Boarded' AND created_at > (now() - interval '6 hours'))
    ) RETURNING id INTO v_halo_id;

    -- 2. DATA DUMP: Move all boarded students into the Halo Manifest
    INSERT INTO public.halo_manifest_snapshots (halo_session_id, student_id)
    SELECT v_halo_id, student_id
    FROM public.student_scan_events
    WHERE bus_id = NEW.bus_id 
    AND event_action = 'Boarded'
    AND created_at > (now() - interval '6 hours');

    -- 3. REQUEST RESCUE BUS (Log the intent in Dispatch)
    INSERT INTO public.incident_dispatch_logs (
        flare_id,
        action_taken,
        dispatch_notes
    ) VALUES (
        NEW.id,
        'HALO_PROTOCOL_INITIATED',
        'Rescue bus requested. Manifest dumped to Pilot Halo App. Initiating headcount sequence.'
    );

    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 4. TRIGGER: Link to EMS Flares (Only for Critical/Bailout types)
DROP TRIGGER IF EXISTS trig_halo_initiate ON public.emergency_flares;
CREATE TRIGGER trig_halo_initiate
AFTER INSERT ON public.emergency_flares
FOR EACH ROW
WHEN (NEW.flare_type = 'Mechanical_Critical' OR NEW.flare_type = 'Crash_Vehicle')
EXECUTE FUNCTION public.fn_initiate_halo_protocol();

COMMIT;

-- ============================================================================
-- ORBIT INTELLIGENCE: FLARE ESCALATION & MANIFEST SNAPSHOT
-- Purpose: Locks student records and creates a manifest during emergencies.
-- Status: Diamond Tier | Safety Critical Logic
-- ============================================================================

BEGIN;

-- 1. THE EMERGENCY SNAPSHOT FUNCTION
CREATE OR REPLACE FUNCTION public.fn_process_flare_escalation()
RETURNS TRIGGER AS $$
DECLARE
    v_student_record RECORD;
BEGIN
    -- Only trigger for high-severity flares
    IF (NEW.severity IN ('Level_2_EMS_Required', 'Level_3_Mass_Casualty')) THEN
        
        -- Create the Dispatch Entry
        INSERT INTO public.incident_dispatch_logs (
            flare_id,
            action_taken,
            dispatch_notes
        ) VALUES (
            NEW.id,
            'SYSTEM: Manifest Snapshot Created',
            'Emergency Flare triggered. All students currently boarded on Bus ' || NEW.bus_id || ' have been flagged for EMS manifest.'
        );

        -- Find every student currently marked as 'Boarded' on this bus
        -- and update their status for first responders.
        -- We use a CTE to ensure we only target the *most recent* scan for each student.
        WITH latest_scans AS (
            SELECT DISTINCT ON (student_id) student_id, event_action
            FROM public.student_scan_events
            WHERE bus_id = NEW.bus_id
            ORDER BY student_id, created_at DESC
        )
        UPDATE public.students
        SET 
            status_note = 'EMERGENCY_MANIFEST_ACTIVE',
            updated_at = clock_timestamp()
        WHERE id IN (SELECT student_id FROM latest_scans WHERE event_action = 'Boarded');

    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 2. THE PANIC BUTTON TRIGGER
DROP TRIGGER IF EXISTS trig_flare_escalation ON public.emergency_flares;
CREATE TRIGGER trig_flare_escalation
AFTER INSERT ON public.emergency_flares
FOR EACH ROW
EXECUTE FUNCTION public.fn_process_flare_escalation();

COMMIT;

-- ============================================================================
-- ORBIT INTELLIGENCE: SMART ETA & MESH ADVANCEMENT
-- Purpose: Automatically advances the trip mesh and updates ETAs for parents.
-- Status: Diamond Tier | Uses PostGIS Spatial Logic
-- ============================================================================

BEGIN;

-- 1. THE MESH ADVANCEMENT FUNCTION
CREATE OR REPLACE FUNCTION public.fn_process_mesh_intelligence()
RETURNS TRIGGER AS $$
DECLARE
    v_trip_id UUID;
    v_current_waypoint_id UUID;
    v_next_waypoint_record RECORD;
    v_geofence_radius INT;
BEGIN
    -- Find the active trip for this bus
    SELECT trip_id INTO v_trip_id 
    FROM public.bus_trip_handshakes 
    WHERE bus_id = NEW.bus_id AND status = 'Active' 
    LIMIT 1;

    IF v_trip_id IS NOT NULL THEN
        -- Get the current target waypoint from the mesh
        SELECT current_waypoint_id INTO v_current_waypoint_id 
        FROM public.trip_active_mesh 
        WHERE trip_id = v_trip_id;

        -- Get the waypoint details (location and radius)
        SELECT location_point, geofence_radius_meters INTO v_next_waypoint_record
        FROM public.route_waypoints 
        WHERE id = v_current_waypoint_id;

        -- SPATIAL CHECK: Is the bus within the geofence? (Using ST_DWithin for speed)
        IF ST_DWithin(NEW.location_gps, v_next_waypoint_record.location_point, v_next_waypoint_record.geofence_radius_meters) THEN
            
            -- ADVANCE THE MESH: Find the next stop in the sequence
            SELECT id INTO v_current_waypoint_id 
            FROM public.route_waypoints 
            WHERE trip_id = v_trip_id 
            AND sequence_index > (SELECT sequence_index FROM public.route_waypoints WHERE id = v_current_waypoint_id)
            ORDER BY sequence_index ASC 
            LIMIT 1;

            -- Update the Active Mesh for the UI to consume
            UPDATE public.trip_active_mesh
            SET 
                current_waypoint_id = v_current_waypoint_id,
                -- In a full rollout, this would call a routing API (like Google/OSRM)
                -- For the Atom, we'll set a placeholder +10 minutes for the next stop
                estimated_arrival_at = clock_timestamp() + interval '10 minutes',
                last_mesh_update = clock_timestamp()
            WHERE trip_id = v_trip_id;

            RAISE NOTICE 'MESH UPDATE: Bus % has arrived at stop. Advancing to next waypoint.', NEW.bus_id;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 2. THE TELEMETRY TRIGGER
-- This fires on every single GPS ping from the Pilot's device.
DROP TRIGGER IF EXISTS trig_mesh_intelligence ON public.bus_telemetry_logs;
CREATE TRIGGER trig_mesh_intelligence
AFTER INSERT ON public.bus_telemetry_logs
FOR EACH ROW
EXECUTE FUNCTION public.fn_process_mesh_intelligence();

COMMIT;

USING (

    AND student_id IN (SELECT id FROM public.students WHERE school_id IN (SELECT id FROM public.schools WHERE district_id = public.get_my_district()))
);
